import fetch from 'node-fetch';

const API_URL = 'https://api-inference.huggingface.co/models/nvidia/ChronoEdit-14B-Diffusers/fal-ai/chrono-edit';

interface HuggingFaceParams {
  imageUrl: string;
  prompt: string;
}

async function query(data: any ) {
  const response = await fetch(API_URL, {
    headers: {
      'Authorization': `Bearer ${process.env.HUGGINGFACE_API_KEY}`,
      'Content-Type': 'application/json',
    },
    method: 'POST',
    body: JSON.stringify(data),
  });
  const result = await response.json();
  return result;
}

export async function editImageWithHuggingFace({ imageUrl, prompt }: HuggingFaceParams): Promise<{ imageUrl: string }> {
  try {
    const response = await query({
      inputs: {
        prompt: prompt,
        image_url: imageUrl,
      },
    });

    if (response && response.image_url) {
      return { imageUrl: response.image_url };
    } else if (response && response.error) {
      console.error('Hugging Face API Error:', response.error);
      throw new Error(`Failed to edit image with Hugging Face: ${response.error}`);
    } else {
      console.error('Unexpected Hugging Face API Response:', response);
      throw new Error('Failed to edit image with Hugging Face: Unexpected response format');
    }
  } catch (error: any) {
    console.error('Error in editImageWithHuggingFace:', error);
    throw new Error(`Failed to edit image with Hugging Face: ${error.message}`);
  }
}
