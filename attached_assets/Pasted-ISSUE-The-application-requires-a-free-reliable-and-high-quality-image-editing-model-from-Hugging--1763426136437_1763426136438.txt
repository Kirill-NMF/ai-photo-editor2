import { HfInference } from "@huggingface/inference";
import fetch from "node-fetch";

// Initialize the Inference Client with your token from Replit Secrets
const hf = new HfInference(process.env.HUGGINGFACE_API_KEY);

interface HuggingFaceParams {
  imageUrl: string;
  prompt: string;
}

export async function editImageWithHuggingFace({ imageUrl, prompt }: HuggingFaceParams): Promise<{ imageUrl: string }> {
  console.log("Attempting to edit image with Qwen/Qwen-Image-Edit model...");

  try {
    // 1. Fetch the image from the URL and get it as a Blob
    console.log(`Fetching image from URL: ${imageUrl}`);
    const imageResponse = await fetch(imageUrl);
    if (!imageResponse.ok) {
      throw new Error(`Failed to fetch image from URL. Status: ${imageResponse.status}`);
    }
    const imageBlob = await imageResponse.blob();
    console.log("Successfully fetched image and converted to Blob.");

    // 2. Call the imageToImage method with the Blob and the chosen model
    const resultBlob = await hf.imageToImage({
      // The best model for our task: instruction-based, popular, and not gated.
      model: "Qwen/Qwen-Image-Edit",
      data: imageBlob,
      parameters: {
        prompt: prompt,
      },
    });
    console.log("Successfully received edited image Blob from Hugging Face.");

    // 3. Convert the resulting Blob to a Base64 Data URL to send to the client
    const arrayBuffer = await resultBlob.arrayBuffer();
    const buffer = Buffer.from(arrayBuffer);
    const base64 = buffer.toString("base64");
    const mimeType = resultBlob.type || "image/jpeg";
    const dataUrl = `data:${mimeType};base64,${base64}`;

    return { imageUrl: dataUrl };

  } catch (error: any) {
    console.error("An exception occurred in editImageWithHuggingFace:", error);
    throw new Error(`Hugging Face SDK Error: ${error.message}`);
  }
}
