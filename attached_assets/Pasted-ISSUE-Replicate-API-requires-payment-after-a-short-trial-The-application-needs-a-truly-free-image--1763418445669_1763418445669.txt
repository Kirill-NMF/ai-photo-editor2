```typescript
import { HfInference } from '@huggingface/inference';
import fetch from 'node-fetch';

/**
 * Edit an image using Hugging Face Inference API (free tier)
 * Requires HUGGINGFACE_API_KEY in environment
 */
export async function editWithHuggingFace(prompt: string, imageUrl: string): Promise<{ imageUrl: string }> {
  const hf = new HfInference(process.env.HUGGINGFACE_API_KEY);

  try {
    if (!imageUrl) {
      throw new Error('Input image is required for Hugging Face image-to-image editing.');
    }

    // Fetch the image from the URL
    const response = await fetch(imageUrl);
    if (!response.ok) {
      throw new Error(`Failed to fetch image from URL: ${imageUrl}`);
    }
    const imageBlob = await response.blob();

    // Call the image-to-image API
    const resultBlob = await hf.imageToImage({
      model: 'Qwen/Qwen-Image-Edit',
      inputs: imageBlob,
      parameters: {
        prompt: prompt,
      },
    });

    // Convert the result blob to a data URL to return to the client
    const arrayBuffer = await resultBlob.arrayBuffer();
    const buffer = Buffer.from(arrayBuffer);
    const dataUrl = `data:${resultBlob.type};base64,${buffer.toString('base64')}`;

    return { imageUrl: dataUrl };
  } catch (error) {
    throw new Error(`Hugging Face API error: ${error.message}`);
  }
}
