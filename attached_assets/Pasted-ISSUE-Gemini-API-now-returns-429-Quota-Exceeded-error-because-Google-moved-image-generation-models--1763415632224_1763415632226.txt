/**
 * Generate an image using Pollinations AI (free, no API key required)
 * Note: Pollinations doesn't support direct image-to-image editing.
 * This function generates a new image based on the prompt.
 */
export async function editWithPollinations(prompt: string, imageUrl?: string): Promise<{ imageUrl: string }> {
  try {
    // Pollinations uses a simple GET request with the prompt in the URL
    const encodedPrompt = encodeURIComponent(prompt);
    const pollinationsUrl = `https://image.pollinations.ai/prompt/${encodedPrompt}`;
    
    // The URL itself is the generated image
    return { imageUrl: pollinationsUrl };
  } catch (error ) {
    throw new Error(`Pollinations API error: ${error.message}`);
  }
}

import Replicate from 'replicate';

/**
 * Generate/edit an image using Replicate API with SDXL model
 * Requires REPLICATE_API_TOKEN in environment
 */
export async function editWithReplicate(prompt: string, imageUrl?: string): Promise<{ imageUrl: string }> {
  const replicate = new Replicate({
    auth: process.env.REPLICATE_API_TOKEN,
  });

  try {
    let output;
    
    if (imageUrl) {
      // Image-to-image editing
      output = await replicate.run(
        "stability-ai/sdxl:39ed52f2a78e934b3ba6e2a89f5b1c712de7dfea535525255b1aa35c5565e08b",
        {
          input: {
            image: imageUrl,
            prompt: prompt,
            refine: "expert_ensemble_refiner",
            scheduler: "K_EULER",
            lora_scale: 0.6,
            num_outputs: 1,
            guidance_scale: 7.5,
            apply_watermark: false,
            high_noise_frac: 0.8,
            negative_prompt: "",
            prompt_strength: 0.8,
            num_inference_steps: 25
          }
        }
      );
    } else {
      // Text-to-image generation
      output = await replicate.run(
        "stability-ai/sdxl:39ed52f2a78e934b3ba6e2a89f5b1c712de7dfea535525255b1aa35c5565e08b",
        {
          input: {
            prompt: prompt,
            num_outputs: 1,
            guidance_scale: 7.5,
            num_inference_steps: 25
          }
        }
      );
    }

    // Replicate returns an array of URLs
    const resultUrl = Array.isArray(output) ? output[0] : output;
    return { imageUrl: resultUrl };
  } catch (error) {
    throw new Error(`Replicate API error: ${error.message}`);
  }
}

// Add these imports at the top
import { editWithPollinations } from './pollinations';
import { editWithReplicate } from './replicate';
import { editWithGemini } from './gemini'; // Ensure this import exists

// Find your existing POST /api/edit endpoint and update it:
app.post('/api/edit', async (req, res) => {
  const { prompt, image, provider } = req.body;

  if (!prompt) {
    return res.status(400).json({ error: 'Prompt is required' });
  }

  if (!provider) {
    return res.status(400).json({ error: 'Provider is required' });
  }

  try {
    let result;
    
    switch (provider) {
      case 'pollinations':
        result = await editWithPollinations(prompt, image);
        break;
      
      case 'replicate':
        if (!process.env.REPLICATE_API_TOKEN) {
          return res.status(500).json({ 
            error: 'Replicate API token not configured. Please add REPLICATE_API_TOKEN to Secrets.' 
          });
        }
        result = await editWithReplicate(prompt, image);
        break;
      
      case 'gemini':
        if (!process.env.GEMINI_API_KEY) {
          return res.status(500).json({ 
            error: 'Gemini API key not configured. Please add GEMINI_API_KEY to Secrets.' 
          });
        }
        result = await editWithGemini(prompt, image);
        break;
      
      default:
        return res.status(400).json({ error: `Invalid provider: ${provider}` });
    }

    res.json(result);
  } catch (error) {
    console.error(`Error with provider ${provider}:`, error);
    res.status(500).json({ 
      error: `Failed to edit image with ${provider}: ${error.message}` 
    });
  }
});
