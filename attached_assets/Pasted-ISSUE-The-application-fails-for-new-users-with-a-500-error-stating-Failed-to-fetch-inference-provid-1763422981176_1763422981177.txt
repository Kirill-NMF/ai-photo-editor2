import fetch from 'node-fetch';

// The full, direct URL to the specific model and provider endpoint.
const API_URL = 'https://api-inference.huggingface.co/models/nvidia/ChronoEdit-14B-Diffusers/fal-ai/chrono-edit';

interface HuggingFaceParams {
  imageUrl: string;
  prompt: string;
}

export async function editImageWithHuggingFace({ imageUrl, prompt }: HuggingFaceParams ): Promise<{ imageUrl: string }> {
  console.log('Attempting to edit image with Hugging Face using direct API call...');

  try {
    const response = await fetch(API_URL, {
      method: 'POST',
      headers: {
        // This is the critical part: Authenticate the request directly.
        'Authorization': `Bearer ${process.env.HUGGINGFACE_API_KEY}`,
        'Content-Type': 'application/json',
      },
      body: JSON.stringify({
        inputs: {
          prompt: prompt,
          // The API expects the image as a publicly accessible URL.
          image_url: imageUrl,
        },
      }),
    });

    if (!response.ok) {
      const errorBody = await response.text();
      console.error(`Hugging Face API responded with status ${response.status}:`, errorBody);
      throw new Error(`API request failed with status ${response.status}: ${errorBody}`);
    }

    const result = await response.json();

    if (result && result.image_url) {
      console.log('Successfully received edited image from Hugging Face.');
      return { imageUrl: result.image_url };
    } else if (result && result.error) {
      console.error('Hugging Face API returned an error:', result.error);
      throw new Error(`Hugging Face API error: ${result.error}`);
    } else {
      console.error('Unexpected response format from Hugging Face:', result);
      throw new Error('Unexpected response format from Hugging Face API.');
    }
  } catch (error: any) {
    console.error('An exception occurred in editImageWithHuggingFace:', error);
    throw new Error(`Failed to edit image with Hugging Face: ${error.message}`);
  }
}
