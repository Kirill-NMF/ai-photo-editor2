ISSUE: Gallery Loading Too Slow - Full-Size Images Causing Performance Issues

Problem Analysis

Current situation:

•
Gallery loads full-size images (5-10 MB each)

•
With 9 projects visible: 45-90 MB total download

•
Loading time on 3G: ~40 seconds

•
Even with lazy loading: still loading large files

Root cause: The gallery displays small preview cards (300x300px approx) but downloads full-resolution images (2000x3000px+).

Example:

Plain Text


Gallery card size: 300x300px
Actual image size: 3000x4000px (7 MB)
Wasted bandwidth: 99% of the image data is not visible!


GOAL

Implement automatic thumbnail generation to create small preview images (300x300px, ~50 KB) that load instantly in the gallery.

Expected improvement:

•
Current: 9 images × 7 MB = 63 MB, ~40 seconds (3G)

•
After: 9 thumbnails × 50 KB = 450 KB, ~3 seconds (3G) ✅ 93% faster!

STRATEGY

Architecture Overview

Plain Text


User uploads image
     ↓
Server receives full image
     ↓
1. Save original (full size) → /uploads/{id}/original.jpg
2. Generate thumbnail (300x300) → /uploads/{id}/thumb.jpg
     ↓
Database stores both URLs:
  - originalUrl: full size (for editor)
  - thumbnailUrl: small (for gallery) ← NEW!
     ↓
Gallery uses thumbnailUrl
Editor uses originalUrl

Memory efficient

Step 3: Create Thumbnail Generation Utility

File: server/thumbnailGenerator.ts (NEW FILE)

TypeScript


import sharp from 'sharp';
import { File } from "@google-cloud/storage";
import { objectStorageClient } from "./objectStorage";
import { randomUUID } from "crypto";

export interface ThumbnailOptions {
  width?: number;
  height?: number;
  quality?: number;
  format?: 'jpeg' | 'webp' | 'png';
}

export class ThumbnailGenerator {
  private defaultOptions: Required<ThumbnailOptions> = {
    width: 300,
    height: 300,
    quality: 80,
    format: 'jpeg',
  };

  /**
   * Generate thumbnail from a Google Cloud Storage file
   * @param sourceFile - Source image file from GCS
   * @param destinationPath - Full GCS path for thumbnail (e.g., "/bucket/uploads/123/thumb.jpg")
   * @param options - Thumbnail generation options
   * @returns Public URL of the generated thumbnail
   */
  async generateThumbnail(
    sourceFile: File,
    destinationPath: string,
    options?: ThumbnailOptions
  ): Promise<string> {
    const opts = { ...this.defaultOptions, ...options };

    try {
      console.log(`[ThumbnailGenerator] Generating thumbnail for ${sourceFile.name}`);
      console.log(`[ThumbnailGenerator] Destination: ${destinationPath}`);

      // Download original image to buffer
      const [imageBuffer] = await sourceFile.download();
      console.log(`[ThumbnailGenerator] Downloaded original (${imageBuffer.length} bytes)`);

      // Generate thumbnail using Sharp
      const thumbnailBuffer = await sharp(imageBuffer)
        .resize(opts.width, opts.height, {
          fit: 'cover',  // Crop to exact dimensions
          position: 'center',  // Center crop
        })
        .toFormat(opts.format, {
          quality: opts.quality,
        })
        .toBuffer();

      console.log(`[ThumbnailGenerator] Generated thumbnail (${thumbnailBuffer.length} bytes)`);
      console.log(`[ThumbnailGenerator] Compression: ${((1 - thumbnailBuffer.length / imageBuffer.length) * 100).toFixed(1)}%`);

      // Parse destination path
      const { bucketName, objectName } = this.parseObjectPath(destinationPath);
      const bucket = objectStorageClient.bucket(bucketName);
      const thumbnailFile = bucket.file(objectName);

      // Upload thumbnail to GCS
      await thumbnailFile.save(thumbnailBuffer, {
        metadata: {
          contentType: `image/${opts.format}`,
          metadata: {
            originalFile: sourceFile.name,
            generatedAt: new Date().toISOString(),
            width: opts.width.toString(),
            height: opts.height.toString(),
          },
        },
        public: true,  // Make thumbnail publicly accessible
      });

      console.log(`[ThumbnailGenerator] Uploaded thumbnail to ${destinationPath}`);

      // Return public URL
      const publicUrl = `https://storage.googleapis.com/${bucketName}/${objectName}`;
      return publicUrl;

    } catch (error ) {
      console.error('[ThumbnailGenerator] Error generating thumbnail:', error);
      throw new Error(`Failed to generate thumbnail: ${error instanceof Error ? error.message : 'Unknown error'}`);
    }
  }

  /**
   * Generate thumbnail from uploaded image during upload flow
   * @param originalFile - Original uploaded file
   * @param uploadId - Unique upload ID
   * @param privateObjectDir - Private object directory from env
   * @returns Thumbnail URL
   */
  async generateUploadThumbnail(
    originalFile: File,
    uploadId: string,
    privateObjectDir: string
  ): Promise<string> {
    // Create thumbnail path: /bucket/uploads/{uploadId}/thumb.jpg
    const thumbnailPath = `${privateObjectDir}/uploads/${uploadId}/thumb.jpg`;
    
    return this.generateThumbnail(originalFile, thumbnailPath, {
      width: 300,
      height: 300,
      quality: 80,
      format: 'jpeg',
    });
  }

  /**
   * Generate thumbnail from edit result
   * @param editResultFile - Edit result file
   * @param editId - Edit ID
   * @param privateObjectDir - Private object directory from env
   * @returns Thumbnail URL
   */
  async generateEditThumbnail(
    editResultFile: File,
    editId: string,
    privateObjectDir: string
  ): Promise<string> {
    const thumbnailPath = `${privateObjectDir}/edits/${editId}/thumb.jpg`;
    
    return this.generateThumbnail(editResultFile, thumbnailPath, {
      width: 300,
      height: 300,
      quality: 80,
      format: 'jpeg',
    });
  }

  private parseObjectPath(path: string): { bucketName: string; objectName: string } {
    if (!path.startsWith("/")) {
      path = `/${path}`;
    }
    const pathParts = path.split("/");
    if (pathParts.length < 3) {
      throw new Error("Invalid path: must contain at least a bucket name");
    }

    const bucketName = pathParts[1];
    const objectName = pathParts.slice(2).join("/");

    return { bucketName, objectName };
  }
}

// Export singleton instance
export const thumbnailGenerator = new ThumbnailGenerator();

Find the image upload endpoint (likely /api/upload or similar) and add thumbnail generation:

import { thumbnailGenerator } from "./thumbnailGenerator";
import { ObjectStorageService } from "./objectStorage";

// ... existing code ...

// Example: Image upload endpoint
app.post("/api/images/upload", async (req, res) => {
  try {
    const { userId } = req.user;  // From auth middleware
    const { uploadUrl, fileName, fileSize, width, height } = req.body;

    // 1. Get the uploaded file from object storage
    const objectStorageService = new ObjectStorageService();
    const uploadedFile = await objectStorageService.getObjectEntityFile(uploadUrl);

    // 2. Generate thumbnail ← NEW!
    const uploadId = extractUploadId(uploadUrl);  // Extract ID from URL
    const privateObjectDir = objectStorageService.getPrivateObjectDir();
    
    console.log('[Upload] Generating thumbnail...');
    const thumbnailUrl = await thumbnailGenerator.generateUploadThumbnail(
      uploadedFile,
      uploadId,
      privateObjectDir
    );
    console.log('[Upload] Thumbnail generated:', thumbnailUrl);

    // 3. Save to database with thumbnail URL
    const image = await db.insert(images).values({
      userId,
      originalUrl: uploadUrl,
      currentUrl: uploadUrl,
      thumbnailUrl,  // ← NEW!
      fileName,
      fileSize,
      width,
      height,
      isOriginal: 1,
    }).returning();

    res.json({ success: true, image: image[0] });

  } catch (error) {
    console.error('[Upload] Error:', error);
    res.status(500).json({ error: 'Failed to upload image' });
  }
});

// Helper function to extract upload ID from URL
function extractUploadId(url: string): string {
  // Example URL: /objects/uploads/abc-123-def/original.jpg
  // Extract: abc-123-def
  const match = url.match(/\/uploads\/([^\/]+)/);
  if (!match) {
    throw new Error('Invalid upload URL format');
  }
  return match[1];
}

Step 6: Backfill Thumbnails for Existing Images (Optional)

Create a migration script to generate thumbnails for existing images.

File: server/scripts/backfillThumbnails.ts (NEW FILE)

TypeScript


import { db } from "../db";
import { images } from "../../shared/schema";
import { thumbnailGenerator } from "../thumbnailGenerator";
import { ObjectStorageService } from "../objectStorage";
import { isNull } from "drizzle-orm";

async function backfillThumbnails() {
  console.log('[Backfill] Starting thumbnail generation for existing images...');

  // Find all images without thumbnails
  const imagesWithoutThumbnails = await db
    .select()
    .from(images)
    .where(isNull(images.thumbnailUrl))
    .limit(100);  // Process in batches

  console.log(`[Backfill] Found ${imagesWithoutThumbnails.length} images without thumbnails`);

  const objectStorageService = new ObjectStorageService();
  const privateObjectDir = objectStorageService.getPrivateObjectDir();

  let successCount = 0;
  let errorCount = 0;

  for (const image of imagesWithoutThumbnails) {
    try {
      console.log(`[Backfill] Processing image ${image.id}...`);

      // Get original file
      const originalFile = await objectStorageService.getObjectEntityFile(image.originalUrl);
      
      // Generate thumbnail
      const uploadId = extractUploadId(image.originalUrl);
      const thumbnailUrl = await thumbnailGenerator.generateUploadThumbnail(
        originalFile,
        uploadId,
        privateObjectDir
      );

      // Update database
      await db
        .update(images)
        .set({ thumbnailUrl })
        .where(eq(images.id, image.id));

      console.log(`[Backfill] ✓ Generated thumbnail for image ${image.id}`);
      successCount++;

    } catch (error) {
      console.error(`[Backfill] ✗ Failed for image ${image.id}:`, error);
      errorCount++;
    }
  }

  console.log(`[Backfill] Complete! Success: ${successCount}, Errors: ${errorCount}`);
}

// Run if called directly
if (require.main === module) {
  backfillThumbnails()
    .then(() => process.exit(0))
    .catch((error) => {
      console.error('[Backfill] Fatal error:', error);
      process.exit(1);
    });
}

function extractUploadId(url: string): string {
  const match = url.match(/\/uploads\/([^\/]+)/);
  if (!match) {
    throw new Error('Invalid upload URL format');
  }
  return match[1];
}
