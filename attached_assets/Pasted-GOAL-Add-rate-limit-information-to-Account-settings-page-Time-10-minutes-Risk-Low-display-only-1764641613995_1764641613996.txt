Add rate limit section:

function AccountPage() {
  const { remaining, resetDate, isAdmin } = useRateLimit();
  
  return (
    <div className="max-w-2xl mx-auto p-6">
      <h1 className="text-2xl font-bold mb-6">Account Settings</h1>
      
      {/* ... existing account info (email, name, etc.) */}
      
      {/* API Usage Section */}
      <div className="mt-8 border-t pt-6">
        <h2 className="text-lg font-semibold mb-4">API Usage</h2>
        
        {isAdmin ? (
          <div className="bg-blue-50 border border-blue-200 rounded-lg p-4">
            <div className="flex items-center gap-2">
              <span className="text-2xl">👑</span>
              <div>
                <p className="font-semibold text-blue-900">Admin Account</p>
                <p className="text-sm text-blue-700">Unlimited AI edits</p>
              </div>
            </div>
          </div>
        ) : (
          <div className="bg-gray-50 border border-gray-200 rounded-lg p-4">
            <div className="flex justify-between items-start mb-3">
              <div>
                <p className="text-sm text-gray-600">AI Edits Remaining</p>
                <p className="text-2xl font-bold text-gray-900">
                  {remaining} <span className="text-base font-normal text-gray-500">/ 11</span>
                </p>
              </div>
              
              <div className="text-right">
                <p className="text-sm text-gray-600">Resets on</p>
                {resetDate && (
                  <p className="text-sm font-semibold text-gray-900">
                    {resetDate.toLocaleDateString('ru-RU', {
                      day: 'numeric',
                      month: 'long',
                      year: 'numeric'
                    })}
                  </p>
                )}
              </div>
            </div>
            
            {/* Progress bar */}
            <div className="w-full bg-gray-200 rounded-full h-2">
              <div
                className="bg-blue-500 h-2 rounded-full transition-all"
                style={{ width: `${(remaining / 11) * 100}%` }}
              />
            </div>
            
            {/* Info text */}
            <p className="text-xs text-gray-500 mt-3">
              Бесплатный тариф включает 11 AI редактирований в месяц. 
              Для увеличения лимита используйте промо-код в Editor.
            </p>
          </div>
        )}
      </div>
    </div>
  );
}
