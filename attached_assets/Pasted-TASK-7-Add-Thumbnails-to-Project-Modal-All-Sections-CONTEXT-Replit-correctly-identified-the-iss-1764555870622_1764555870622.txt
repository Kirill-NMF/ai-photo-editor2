/**
 * Generate thumbnail for AI edit result (not saved to gallery)
 * @param editId - Database ID of the edit
 * @param resultUrl - Object path of the AI edit result
 */
export async function generateEditThumbnailInBackground(
  editId: number,
  resultUrl: string
): Promise<void> {
  try {
    console.log(`[EditThumbnailBG] Starting thumbnail generation for edit ${editId}`);
    console.log(`[EditThumbnailBG] Result URL: ${resultUrl}`);

    // Initialize object storage service
    const objectStorageService = new ObjectStorageService();
    const privateObjectDir = objectStorageService.getPrivateObjectDir();

    // Get the result file from GCS
    const resultFile = await objectStorageService.getObjectEntityFile(resultUrl);
    
    // Extract edit ID from the path
    const editIdFromPath = extractEditIdFromPath(resultUrl);
    console.log(`[EditThumbnailBG] Extracted edit ID: ${editIdFromPath}`);

    // Generate WebP thumbnail
    const startTime = Date.now();
    const thumbnailUrl = await thumbnailGenerator.generateEditThumbnail(
      resultFile,
      editIdFromPath,
      privateObjectDir
    );
    const duration = Date.now() - startTime;
    
    console.log(`[EditThumbnailBG] Thumbnail generated in ${duration}ms: ${thumbnailUrl}`);

    // Update edits table with thumbnail URL
    await db
      .update(edits)
      .set({ thumbnailUrl })
      .where(eq(edits.id, editId));

    console.log(`[EditThumbnailBG] ✓ Database updated for edit ${editId}`);

  } catch (error) {
    console.error(`[EditThumbnailBG] ✗ Failed to generate thumbnail for edit ${editId}:`, error);
    console.error(`[EditThumbnailBG] Result URL: ${resultUrl}`);
  }
}

/**
 * Extract edit ID from object path
 * Handles: /objects/edits/def-456/result.jpg → def-456
 */
function extractEditIdFromPath(objectPath: string): string {
  const parts = objectPath.split('/');
  const editsIndex = parts.indexOf('edits');
  
  if (editsIndex !== -1 && editsIndex + 1 < parts.length) {
    return parts[editsIndex + 1];
  }
  
  throw new Error(`Cannot extract edit ID from path: ${objectPath}`);
}
