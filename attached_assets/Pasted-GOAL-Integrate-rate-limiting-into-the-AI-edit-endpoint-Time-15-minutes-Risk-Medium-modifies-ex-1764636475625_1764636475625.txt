Step 3: Integrate rate limiting logic

Replace the edit endpoint with:

TypeScript


app.post("/api/edits", async (req, res) => {
  if (!req.user) {
    return res.status(401).json({ error: "Unauthorized" });
  }

  const { imageId, prompt, baseEditId } = req.body;
  
  // === PROMO CODE CHECK ===
  if (isPromoCode(prompt)) {
    const result = await applyPromoCode(req.user.id);
    
    if (result.success) {
      return res.json({
        success: true,
        isPromoCode: true,
        message: result.message,
      });
    } else {
      return res.status(400).json({
        success: false,
        isPromoCode: true,
        error: result.message,
      });
    }
  }
  
  // === RATE LIMIT CHECK ===
  const rateLimit = await checkRateLimit(req.user.id);
  
  if (!rateLimit.allowed) {
    return res.status(429).json({
      error: "API limit reached",
      remaining: 0,
      resetDate: rateLimit.resetDate,
      limitReached: true,
    });
  }
  
  // === EXISTING VALIDATION ===
  // ... your existing validation code
  
  try {
    // === CALL AI API ===
    // ... your existing AI generation code
    const result = await generateEdit(prompt, imageUrl);
    
    // === INCREMENT COUNTER (only on success) ===
    await incrementRateLimit(req.user.id);
    
    // === SAVE TO DATABASE ===
    // ... your existing database save code
    const edit = await db.insert(edits).values({...});
    
    // === GET UPDATED RATE LIMIT ===
    const newRateLimit = await checkRateLimit(req.user.id);
    
    // === RETURN SUCCESS WITH RATE LIMIT INFO ===
    return res.json({
      success: true,
      edit: savedEdit,
      rateLimit: {
        remaining: newRateLimit.remaining,
        isLastEdit: newRateLimit.isLastEdit,
        resetDate: newRateLimit.resetDate,
      },
    });
    
  } catch (error) {
    // Don't increment counter if edit failed
    console.error('[Edit] Failed:', error);
    return res.status(500).json({ error: "Edit generation failed" });
  }
});


Key changes:

1.
Check if prompt is promo code → apply and return

2.
Check rate limit → block if exceeded

3.
Call AI API (existing code)

4.
Increment counter only on success

5.
Return rate limit info in response




Step 4: Add rate limit status endpoint

Add new endpoint in server/routes.ts:

TypeScript


// Get current rate limit status
app.get("/api/rate-limit", async (req, res) => {
  if (!req.user) {
    return res.status(401).json({ error: "Unauthorized" });
  }

  const rateLimit = await checkRateLimit(req.user.id);
  
  return res.json({
    remaining: rateLimit.remaining,
    resetDate: rateLimit.resetDate,
    isAdmin: rateLimit.remaining === 999,
  });
});
