import { db } from "./db";
import { users } from "../shared/schema";
import { eq } from "drizzle-orm";

export const API_LIMIT_FREE = 11;
export const PROMO_CODE = "вот тебе пряник"; // Lowercase for case-insensitive check

export interface RateLimitResult {
  allowed: boolean;
  remaining: number;
  isLastEdit: boolean; // True if this is the 10th edit (show warning)
  resetDate: Date;
}

/**
 * Check if user can make an API request
 */
export async function checkRateLimit(userId: string): Promise<RateLimitResult> {
  const user = await db.select().from(users).where(eq(users.id, userId)).limit(1);
  
  if (!user || user.length === 0) {
    throw new Error("User not found");
  }

  const userData = user[0];

  // Admin has unlimited access
  if (userData.isAdmin === 1) {
    return {
      allowed: true,
      remaining: 999,
      isLastEdit: false,
      resetDate: new Date(),
    };
  }

  // Check if we need to reset counter (monthly reset)
  const now = new Date();
  let resetDate = userData.apiRequestsResetDate;

  if (!resetDate) {
    // First time - set reset date to next month same day
    resetDate = new Date(now);
    resetDate.setMonth(resetDate.getMonth() + 1);
    
    await db.update(users)
      .set({ apiRequestsResetDate: resetDate })
      .where(eq(users.id, userId));
  } else if (now >= resetDate) {
    // Reset counter and set next reset date
    const nextResetDate = new Date(resetDate);
    nextResetDate.setMonth(nextResetDate.getMonth() + 1);
    
    await db.update(users)
      .set({
        apiRequestsCount: 0,
        apiRequestsResetDate: nextResetDate,
      })
      .where(eq(users.id, userId));
    
    resetDate = nextResetDate;
    userData.apiRequestsCount = 0;
  }

  const currentCount = userData.apiRequestsCount || 0;
  const limit = API_LIMIT_FREE;
  const remaining = Math.max(0, limit - currentCount);

  return {
    allowed: currentCount < limit,
    remaining,
    isLastEdit: currentCount === limit - 1, // 10th edit (before last)
    resetDate,
  };
}

/**
 * Increment user's API request counter
 * Call this AFTER successful edit
 */
export async function incrementRateLimit(userId: string): Promise<void> {
  await db.update(users)
    .set({
      apiRequestsCount: db.raw(`api_requests_count + 1`),
    })
    .where(eq(users.id, userId));
}

/**
 * Check if prompt contains promo code
 */
export function isPromoCode(prompt: string): boolean {
  return prompt.trim().toLowerCase() === PROMO_CODE;
}

/**
 * Apply promo code - reset counter and mark as used
 */
export async function applyPromoCode(userId: string): Promise<{ success: boolean; message: string }> {
  const user = await db.select().from(users).where(eq(users.id, userId)).limit(1);
  
  if (!user || user.length === 0) {
    return { success: false, message: "User not found" };
  }

  const userData = user[0];

  // Check if promo code already used
  if (userData.promoCodeUsed === 1) {
    return {
      success: false,
      message: "Промо-код уже был использован ранее",
    };
  }

  // Reset counter and mark promo code as used
  await db.update(users)
    .set({
      apiRequestsCount: 0,
      promoCodeUsed: 1,
    })
    .where(eq(users.id, userId));

  return {
    success: true,
    message: "Спасибо за пряник. Теперь я сыт и могу служить дальше",
  };
}
